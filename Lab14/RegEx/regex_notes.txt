r"abc"      => "abc"
r"[abc]"    => "a", "b", "c"
r"[^abc]"   => "~", "\n", .... any symbol, but not a or b or c
r"[+.]"     =>  "+", "."
r"[a-z]"    => "a", "b", ...., "z"

r"[0-9]"    => 0, 1,2, ...9
r"\d"       => 0, 1,2, ...9

r"."        => '!', 'a' , any symbol except '\n'

r"a+"       => 'aaaaaaa...' ,'aaaaa', ....,'a'

r"a?"      => "a", ""
r"a??"     => "", "a"


r"1^"     => no match ever
r"1$"     => "131", "1", ...
r"^$"     => ""


r"cat|dog" => "cat" or "dog",
